 blocks and any other whitespace-sensitive HTML. --}}

{{#if (eq frontmatter.template "cover")}}
{{> cover this}}
{{else if (eq frontmatter.template "back-cover")}}
{{> back-cover this}}
{{else if (eq frontmatter.template "plain")}}
{{> plain this}}
{{else}}
{{> chapter this}}
{{/if}}
{{/each}}
